<script>
const STORAGE_KEY = 'bgt_v1';
const CATS = ['Rent / Housing','Food','Groceries','Transportation','Utilities','Entertainment','Subscriptions','Shopping','Healthcare','Other'];
const CAT_COLORS = ['#34d399','#60a5fa','#a78bfa','#f97316','#facc15','#f472b6','#38bdf8','#fb7185','#4ade80','#94a3b8'];

let data = [];
let filter = 'month';
let activeForm = 'income';
let pieChart = null;

function save() { localStorage.setItem(STORAGE_KEY, JSON.stringify(data)); }
function load() { try { data = JSON.parse(localStorage.getItem(STORAGE_KEY)) || []; } catch { data = []; } }

function today() { return new Date().toISOString().slice(0,10); }

function resetAll() {
  if (!confirm('Delete ALL data and start fresh? This cannot be undone.')) return;
  data = []; save(); render(); toast('All data cleared.');
}

function switchForm(type) {
  activeForm = type;
  document.getElementById('form-income').classList.toggle('hidden', type !== 'income');
  document.getElementById('form-expense').classList.toggle('hidden', type !== 'expense');
  document.getElementById('tab-income').classList.toggle('active', type === 'income');
  document.getElementById('tab-expense').classList.toggle('active', type === 'expense');
}

function setFilter(f) {
  filter = f;
  document.getElementById('filter-month').classList.toggle('active', f === 'month');
  document.getElementById('filter-all').classList.toggle('active', f === 'all');
  render();
}

function addTransaction(type) {
  const amt = parseFloat(type === 'income' ? document.getElementById('inc-amount').value : document.getElementById('exp-amount').value);
  if (!amt || amt <= 0) { toast('Enter a valid amount.'); return; }

  if (type === 'income') {
    const desc = document.getElementById('inc-desc').value.trim() || 'Income';
    const date = document.getElementById('inc-date').value || today();
    data.push({ id: Date.now(), type: 'income', amount: amt, desc, date });
    document.getElementById('inc-amount').value = '';
    document.getElementById('inc-desc').value = '';
    document.getElementById('inc-date').value = '';
    toast('Income added ✓');
  } else {
    const cat = document.getElementById('exp-cat').value;
    if (!cat) { toast('Select a category.'); return; }
    const desc = document.getElementById('exp-desc').value.trim();
    const date = document.getElementById('exp-date').value || today();
    data.push({ id: Date.now(), type: 'expense', amount: amt, cat, desc, date });
    document.getElementById('exp-amount').value = '';
    document.getElementById('exp-desc').value = '';
    document.getElementById('exp-date').value = '';
    document.getElementById('exp-cat').value = '';
    toast('Expense added ✓');
  }
  save(); render();
}

function deleteTransaction(id) {
  data = data.filter(t => t.id !== id);
  save(); render(); toast('Deleted.');
}

function filtered() {
  if (filter === 'all') return data;
  const now = new Date();
  const y = now.getFullYear(), m = now.getMonth();
  return data.filter(t => {
    const d = new Date(t.date);
    return d.getFullYear() === y && d.getMonth() === m;
  });
}

function fmt(n) { return '$' + n.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 }); }

function render() {
  const tx = filtered().slice().sort((a,b) => b.date.localeCompare(a.date) || b.id - a.id);
  const income = tx.filter(t => t.type === 'income').reduce((s,t) => s + t.amount, 0);
  const expenses = tx.filter(t => t.type === 'expense').reduce((s,t) => s + t.amount, 0);
  const balance = income - expenses;
  const savingsRate = income > 0 ? Math.round(((income - expenses) / income) * 100) : null;

  document.getElementById('d-income').textContent = fmt(income);
  document.getElementById('d-expenses').textContent = fmt(expenses);
  const balEl = document.getElementById('d-balance');
  balEl.textContent = fmt(balance);
  balEl.className = 'syne text-4xl font-800 ' + (balance >= 0 ? 'text-emerald-400 balance-glow' : 'text-red-400 deficit-glow');
  document.getElementById('d-savings').textContent = savingsRate !== null ? savingsRate + '%' : '—';
  document.getElementById('d-savings').className = 'syne text-2xl font-700 ' + (savingsRate !== null ? (savingsRate >= 20 ? 'text-emerald-400' : savingsRate >= 0 ? 'text-yellow-400' : 'text-red-400') : 'text-zinc-500');
  document.getElementById('d-count').textContent = tx.length;

  // Transaction list
  const list = document.getElementById('tx-list');
  const empty = document.getElementById('tx-empty');
  if (tx.length === 0) { list.innerHTML = ''; empty.classList.remove('hidden'); }
  else {
    empty.classList.add('hidden');
    list.innerHTML = tx.map(t => {
      const isIncome = t.type === 'income';
      const catColor = isIncome ? '#34d399' : CAT_COLORS[CATS.indexOf(t.cat)] || '#94a3b8';
      const label = isIncome ? 'Income' : t.cat;
      return `<div class="tx-row flex items-center gap-3 py-3 px-1 fade-in">
        <div class="flex-shrink-0">
          <span class="cat-dot" style="background:${catColor};"></span>
        </div>
        <div class="flex-1 min-w-0">
          <div class="text-sm text-zinc-200 truncate">${escHtml(t.desc || label)}</div>
          <div class="text-xs text-zinc-600 mt-0.5 flex gap-2">
            <span>${label}</span>
            <span>·</span>
            <span>${fmtDate(t.date)}</span>
          </div>
        </div>
        <div class="text-sm font-500 flex-shrink-0 ${isIncome ? 'text-emerald-400' : 'text-red-400'}">
          ${isIncome ? '+' : '-'}${fmt(t.amount)}
        </div>
        <button class="btn-danger flex-shrink-0" onclick="deleteTransaction(${t.id})">del</button>
      </div>`;
    }).join('');
  }

  // Pie chart
  renderChart(tx);
}

function fmtDate(d) {
  const dt = new Date(d + 'T00:00:00');
  return dt.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
}

function escHtml(s) {
  return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function renderChart(tx) {
  const expenses = tx.filter(t => t.type === 'expense');
  const totals = {};
  expenses.forEach(t => { totals[t.cat] = (totals[t.cat] || 0) + t.amount; });
  const labels = Object.keys(totals);
  const values = Object.values(totals);
  const colors = labels.map(l => CAT_COLORS[CATS.indexOf(l)] || '#94a3b8');

  if (pieChart) { pieChart.destroy(); pieChart = null; }

  const canvas = document.getElementById('pieChart');
  if (labels.length === 0) {
    canvas.parentElement.style.height = '60px';
    document.getElementById('chart-legend').innerHTML = '<div class="text-xs text-zinc-600 text-center">no expense data</div>';
    return;
  }
  canvas.parentElement.style.height = '200px';

  pieChart = new Chart(canvas, {
    type: 'doughnut',
    data: {
      labels,
      datasets: [{ data: values, backgroundColor: colors, borderWidth: 0, hoverOffset: 4 }]
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      cutout: '62%',
      plugins: {
        legend: { display: false },
        tooltip: {
          callbacks: {
            label: ctx => ` ${fmt(ctx.raw)} (${Math.round(ctx.raw / values.reduce((a,b)=>a+b,0)*100)}%)`
          },
          bodyFont: { family: 'DM Mono', size: 12 },
          backgroundColor: '#1f1f23',
          borderColor: '#27272a',
          borderWidth: 1,
          padding: 10,
        }
      }
    }
  });

  const total = values.reduce((a,b)=>a+b,0);
  document.getElementById('chart-legend').innerHTML = labels.map((l,i) => `
    <div class="flex items-center justify-between text-xs">
      <span class="flex items-center gap-2 text-zinc-400">
        <span class="cat-dot" style="background:${colors[i]};"></span>${l}
      </span>
      <span class="text-zinc-500">${fmt(values[i])} <span class="text-zinc-700">·</span> ${Math.round(values[i]/total*100)}%</span>
    </div>
  `).join('');
}

let toastTimer;
function toast(msg) {
  const el = document.getElementById('toast');
  el.textContent = msg;
  el.classList.remove('hidden');
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => el.classList.add('hidden'), 2200);
}

// Init date fields
document.getElementById('inc-date').value = today();
document.getElementById('exp-date').value = today();

load();
render();
</script>
